\begin{frame}{Le cours \gloss{The course}}
  \small
  \only<1>{
    Goals:
    \begin{enumerate}
      \item Gain some competence in speaking, understanding, reading, and writing Louisiana French
      \item Have fun
    \end{enumerate}
    What to expect:
    \begin{enumerate}
      \item No grades, no homework (basically)
      \item Vocabulary lists
      \item Grammar lessons in class with optional grammar exercises at home
      \item Lots and lots of practice
      \item Lots and lots of French in class
    \end{enumerate}
  }
