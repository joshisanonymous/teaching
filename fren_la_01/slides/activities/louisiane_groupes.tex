        Un groupe changeant \gloss{An inconstant group}
        \begin{enumerate}
          \item Early colonial $\to$ Anyone and anything native-born
          \item Early 1800s $\to$ Les gens de couleur libres \gloss{Free people of color}
          \item Late 1800s $\to$ Any White person with French or Spanish colonial ancestors
          \item 1900s onward $\to$ Any Black person with roots in South Louisiana
        \end{enumerate}
    }
